43template<typename JT, typename FT, typename ST> requires is_arma_mat<double, JT> && is_arma_mat<double, FT> && std::invocable<ST, const FT&> && std::same_as<std::invoke_result_t<ST, const FT&>, std::pair<FT, JT>> int homotopy_solve(FT& x, ST&& system, const homotopy_config& config) {
44 const auto initial_f = system(x).first;
45
46 auto counter{0u};
47 auto current_t{0.};
48 auto current_x = x;
49 auto incre_t = config.incre_t;
50
51 while(current_t < 1.) {
52 const auto trial_t = std::min(1., current_t + incre_t);
53 const FT target_f = (1. - trial_t) * initial_f;
54 auto trial_x = current_x;
55
56 FT incre_x, residual;
57 JT jacobian;
58
59 const auto bounding_eval = [&](const FT& in_x) {
60 if(++counter >= config.max_evaluation) return false;
61 std::tie(residual, jacobian) = system(in_x);
62 residual -= target_f;
63 return true;
64 };
65
66 auto ref_error{1.};
67 auto converged{false};
68
69 for(auto round{0u}; round < config.max_iteration; ++round) {
70 if(!bounding_eval(trial_x)) return SUANPAN_FAIL;
71
72 if(!solve(incre_x, jacobian, residual, solve_opts::equilibrate + solve_opts::no_approx)) break;
73 trial_x -= incre_x;
74
75 const auto error = suanpan::inf_norm(incre_x);
76 if(0u == round) ref_error = error;
77 suanpan_debug("Homotopy iteration error: {:.5E} at progress {:.3f}.\n", error, trial_t);
78
79 if(error < config.tolerance * ref_error || ((error < config.tolerance || suanpan::inf_norm(residual) < config.tolerance) && round > 5u)) {
80 current_t = trial_t;
81 current_x = trial_x;
82
83 if(!config.fixed_incre && round <= config.scaling_counter) incre_t *= 1.5;
84
85 converged = true;
86
87 break;
88 }
89 }
90
91 if(!converged && (config.fixed_incre || (incre_t *= .5) < config.min_incre)) return SUANPAN_FAIL;
92 }
93
94 x = current_x;
95
96 return SUANPAN_SUCCESS;
97}
