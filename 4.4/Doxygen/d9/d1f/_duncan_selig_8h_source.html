63 template<bool plastic, bool with_jacobian> [[nodiscard]] auto local_assemble(const vec& t_stress) {
64 const auto [elastic, bulk, deds, dkds] = compute_moduli<plastic>(t_stress);
65
66 const auto factor_a = 3. * bulk * (3. * bulk + elastic);
67 const auto factor_b = 3. * bulk * (3. * bulk - elastic);
68
69 mat33 right(fill::zeros);
70 right(0, 0) = right(1, 1) = factor_a;
71 right(0, 1) = right(1, 0) = factor_b;
72 right(2, 2) = 3. * bulk * elastic;
73
74 const vec3 stress_diff = t_stress - current_stress;
75 const vec3 residual = (9. * bulk - elastic) * stress_diff - right * incre_strain;
76
77 mat33 jacobian;
78
79 if constexpr(with_jacobian) {
80 const rowvec3 factor_c = 3. * (elastic * dkds + bulk * deds);
81 const rowvec3 dfads = 18. * bulk * dkds + factor_c;
82 const rowvec3 dfbds = 18. * bulk * dkds - factor_c;
83
84 jacobian = (9. * bulk - elastic) * eye(3, 3) + stress_diff * (9. * dkds - deds);
85 jacobian.row(0) -= incre_strain(0) * dfads + incre_strain(1) * dfbds;
86 jacobian.row(1) -= incre_strain(0) * dfbds + incre_strain(1) * dfads;
87 jacobian.row(2) -= incre_strain(2) * factor_c;
88 }
89
90 return std::tuple{residual, right, jacobian};
91 }

93 template<bool plastic> int local_update() {
94 auto ref_error = 0.;
95 vec3 incre;
96
97 auto counter = 0u;
98 while(true) {
99 if(max_iteration < ++counter) {
100 suanpan_debug("Local iteration cannot converge within {} iterations.\n", max_iteration);
101 return SUANPAN_FAIL;
102 }
103
104 const auto [residual, right, jacobian] = local_assemble<plastic, true>(trial_stress);
105
106 if(!solve(incre, jacobian, residual)) return SUANPAN_FAIL;
107
108 const auto error = suanpan::inf_norm(incre);
109 if(1u == counter) ref_error = error;
110 suanpan_debug("Local iteration error: {:.5E}.\n", error);
111 const auto residual_norm = suanpan::inf_norm(residual);
112 if(error < tolerance * ref_error || ((error < tolerance || residual_norm < tolerance) && counter > 5u)) {
113 if(!solve(trial_stiffness, jacobian, right)) return SUANPAN_FAIL;
114
115 return SUANPAN_SUCCESS;
116 }
117
118 // backtracking line search on the residual norm
119 auto step = 1.;
120 vec candidate = trial_stress - incre;
121 while(step > 1E-3 && suanpan::inf_norm(std::get<0>(local_assemble<plastic, false>(candidate))) >= residual_norm) candidate = trial_stress - (step *= .5) * incre;
122
123 trial_stress = candidate;
124 }
125 }
