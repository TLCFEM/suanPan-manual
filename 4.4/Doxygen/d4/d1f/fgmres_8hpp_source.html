62int fgmres_solve(const T& kernel, const vec& diagonal, double* left, double* right, const double tolerance, MKL_INT* itercount = nullptr) {
63 const auto n = static_cast<MKL_INT>(kernel.n_rows);
64 const auto r = std::min(MKL_INT{150}, n);
65
66 if(n <= 0 || nullptr == left || nullptr == right || !std::isfinite(tolerance) || tolerance <= 0. || diagonal.n_elem != static_cast<uword>(n)) {
67 suanpan_error("Invalid arguments.\n");
68 return -1;
69 }
70
71 for(const auto value : diagonal)
72 if(0. == value || !std::isfinite(value)) {
73 suanpan_error("Preconditioner diagonal must be finite and nonzero.\n");
74 return -1;
75 }
76
77 // oneMKL requires ((2*r + 1)*n + r*(r + 9)/2 + 1) double slots in tmp.
78 std::vector work(static_cast<std::size_t>((2 * r + 1) * n + r * (r + 9) / 2 + 1), 0.);
79 std::array<MKL_INT, 128> ipar{};
80 std::array<double, 128> dpar{};
81 MKL_INT info{};
82
83 // Pass x and b as well: a non-null initial guess makes the call well-defined.
84 dfgmres_init(&n, left, right, &info, ipar.data(), dpar.data(), work.data());
85 if(0 != info) {
86 suanpan_error("Initialization failed with code {}.\n", info);
87 return -2;
88 }
89
90 ipar[8] = 1; // residual stopping test
91 ipar[9] = 0; // no user-defined stopping test
92 ipar[10] = 1; // use preconditioner
93 ipar[11] = 1; // automatic zero-norm test
94 ipar[12] = 0; // write the solution into x and keep b untouched
95 dpar[0] = tolerance;
96
97 // Recommended after changing ipar/dpar; it also fixes ipar[15:20] if needed.
98 dfgmres_check(&n, left, right, &info, ipar.data(), dpar.data(), work.data());
99 if(0 != info) {
100 // -1001/-1010/-1011 only report warnings or adjusted parameters.
101 if(info < -1011) {
102 suanpan_error("Parameter check failed with code {}.\n", info);
103 return -3;
104 }
105 suanpan_debug("Parameter check reported code {}.\n", info);
106 }
107
108 const auto limit = std::max(MKL_INT{32}, 10 * ipar[4] + 32);
109 auto converged{false};
110 auto exhausted{false};
111
112 for(MKL_INT count = 0; count < limit && !converged && !exhausted; ++count) {
113 dfgmres(&n, left, right, &info, ipar.data(), dpar.data(), work.data());
114
115 switch(info) {
116 case 0:
117 converged = true;
118 break;
119 case 1: // y = A*x
120 case 3: { // y = M^-1*x
121 if(ipar[21] < 1 || ipar[22] < 1 || static_cast<std::size_t>(ipar[21] + n - 1) > work.size() || static_cast<std::size_t>(ipar[22] + n - 1) > work.size()) {
122 suanpan_error("Invalid work offsets returned by oneMKL.\n");
123 return -4;
124 }
125 const vec xn(&work[ipar[21] - 1], static_cast<uword>(n), false, true);
126 vec yn(&work[ipar[22] - 1], static_cast<uword>(n), false, true);
127 if(1 == info) {
128 const mat product = kernel * xn;
129 if(product.n_rows != static_cast<uword>(n) || product.n_cols != 1) {
130 suanpan_error("Matrix-vector product has unexpected size.\n");
131 return -4;
132 }
133 yn = product;
134 }
135 else yn = xn / diagonal;
136 break;
137 }
138 case 2: // user stopping test, normally disabled
139 if(dpar[4] <= dpar[3]) converged = true;
140 break;
141 case 4: // zero-norm check, compare with dpar[7], not the residual tolerance
142 if(dpar[6] <= dpar[7]) converged = true;
143 break;
144 case -1: // iteration limit reached before convergence
145 exhausted = true;
146 break;
147 default:
148 suanpan_error("Solver returned error code {}.\n", info);
149 return -4;
150 }
151 }
152
153 if(!converged && !exhausted) {
154 suanpan_error("Exceeded {} reverse-communication requests.\n", limit);
155 return -5;
156 }
157
158 MKL_INT counter{};
159 dfgmres_get(&n, left, right, &info, ipar.data(), dpar.data(), work.data(), &counter);
160 if(nullptr != itercount) *itercount = counter;
161 if(0 != info) {
162 suanpan_error("Could not retrieve the solution (code {}).\n", info);
163 return -6;
164 }
165
166 for(uword i = 0; i < static_cast<uword>(n); ++i)
167 if(!std::isfinite(left[i])) {
168 suanpan_error("Solution contains non-finite values.\n");
169 return -4;
170 }
171
172 if(!converged) {
173 suanpan_debug("Stopped after {} iterations without convergence.\n", counter);
174 return -7;
175 }
176
177 suanpan_debug("Converged in {} iterations.\n", counter);
178 return 0;
179}
