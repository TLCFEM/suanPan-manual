83template<sp_d T> int BandMatSpikeGPU<T>::direct_solve(Mat<T>& X, Mat<T>&& B) {
84 suanpan_assert([&] { if(this->n_rows != this->n_cols) throw std::invalid_argument("requires a square matrix"); });
85
86 const auto N = static_cast<int>(this->n_rows);
87 const auto KL = static_cast<int>(this->l_band);
88 const auto KU = static_cast<int>(this->u_band);
89 const auto NRHS = static_cast<int>(B.n_cols);
90 const auto LDAB = static_cast<int>(this->m_rows);
91 const auto LDB = static_cast<int>(B.n_rows);
92
93 if(!this->factored) {
94 solver.reset();
95 mixer.reset();
96
97 d_A.copy_from(this->memptr());
98
99 if constexpr(std::is_same_v<T, float>) solver = std::make_unique<gspike::solver<float>>(N, KL, KU, d_A.get<float>(), LDAB);
100 else mixer = std::make_unique<gspike::mixed_solver>(N, KL, KU, d_A.get<double>(), LDAB);
101
102 this->factored = true;
103 }
104
105 // sync copies transfer the whole buffer, so it must match the RHS exactly
106 if(d_B.size != static_cast<size_t>(B.n_elem)) d_B = cuda_ptr(sizeof(T), static_cast<size_t>(B.n_elem));
107
108 d_B.copy_from(B.memptr());
109
110 auto INFO = 0;
111 if constexpr(std::is_same_v<T, float>) INFO = solver->solve(NRHS, d_B.get<float>(), LDB);
112 else INFO = mixer->solve(NRHS, d_B.get<double>(), LDB);
113
114 if(0 != INFO) {
115 suanpan_error("Error code {} received, the matrix is probably singular.\n", INFO);
116 return INFO;
117 }
118
119 d_B.copy_to(B.memptr());
120
121 X = std::move(B);
122
123 return SUANPAN_SUCCESS;
124}
