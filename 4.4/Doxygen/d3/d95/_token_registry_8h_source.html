41template<typename V> class TokenRegistry {
42 std::unordered_map<std::string, V> pool;
43
44public:
45 TokenRegistry(std::initializer_list<std::pair<std::string_view, V>> entries) {
46 pool.reserve(entries.size());
47 for(const auto& [token, value] : entries) pool.emplace(suanpan::to_lower(token), value);
48 }
49
50 [[nodiscard]] V fetch(const std::string_view token) const {
51 if(const auto iter = pool.find(suanpan::to_lower(token)); iter != pool.cend()) return iter->second;
52 return V{};
53 }
54
55 [[nodiscard]] auto size() const { return pool.size(); }
56};
